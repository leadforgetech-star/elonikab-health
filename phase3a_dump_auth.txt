  28  def _is_https(request):
  29      """Detecte si la requete arrive via HTTPS (direct ou via tunnel/proxy)."""
  30      try:
  31          if request.url.scheme == "https":
  32              return True
  33          # Cloudflare envoie X-Forwarded-Proto
  34          proto = request.headers.get("x-forwarded-proto", "")
  35          if proto.lower() == "https":
  36              return True
  37          # Header Cloudflare specifique
  38          cf_visitor = request.headers.get("cf-visitor", "")
  39          if "https" in cf_visitor.lower():
  40              return True
  41      except Exception:
  42          pass
  43      return False

  62  def hash_password(password):
  63      if not password or len(password) < 6:
  64          raise ValueError("Mot de passe trop court (6 caracteres minimum)")
  65      salt = secrets.token_bytes(16)
  66      digest = hashlib.scrypt(password.encode("utf-8"), salt=salt,
  67                              n=2**14, r=8, p=1, dklen=32)
  68      return "scrypt$" + base64.b64encode(salt).decode() + "$" + base64.b64encode(digest).decode()

  71  def verify_password(password, stored):
  72      try:
  73          scheme, salt_b64, digest_b64 = stored.split("$")
  74          if scheme != "scrypt":
  75              return False
  76          salt = base64.b64decode(salt_b64)
  77          expected = base64.b64decode(digest_b64)
  78          actual = hashlib.scrypt(password.encode("utf-8"), salt=salt,
  79                                  n=2**14, r=8, p=1, dklen=32)
  80          return hmac.compare_digest(expected, actual)
  81      except Exception:
  82          return False

 136      def _create_session(conn, user_id):
 137          token = new_token()
 138          now = datetime.now()
 139          exp = now + timedelta(days=SESSION_TTL_DAYS)
 140          conn.execute(
 141              "INSERT INTO sessions (token, user_id, created_at, expires_at)"
 142              " VALUES (?, ?, ?, ?)",
 143              (token, user_id,
 144               now.isoformat(timespec="seconds"),
 145               exp.isoformat(timespec="seconds")),
 146          )
 147          return token

 149      def load_user_from_cookie(session_token):
 150          if not session_token:
 151              return None
 152          conn = get_db()
 153          try:
 154              row = conn.execute(
 155                  "SELECT u.* FROM sessions s JOIN users u ON u.id = s.user_id"
 156                  " WHERE s.token = ? AND s.expires_at > ?",
 157                  (session_token, datetime.now().isoformat(timespec="seconds")),
 158              ).fetchone()
 159              return row
 160          finally:
 161              conn.close()

 169      def require_role(*roles):
 170          def dep(session_token: Optional[str] = Cookie(None, alias=SESSION_COOKIE)):
 171              row = load_user_from_cookie(session_token)
 172              if row is None:
 173                  raise HTTPException(401, "Non authentifie")
 174              if not row["is_active"]:
 175                  raise HTTPException(403, "Compte desactive")
 176              if row["role"] not in roles:
 177                  raise HTTPException(403, "Acces refuse pour ce role")
 178              return _public(row)
 179          return dep

 298      @app.patch("/api/auth/me")
 299      def update_me(payload: UpdateProfileRequest,
 300                    session_token: Optional[str] = Cookie(None, alias=SESSION_COOKIE)):
 301          row = load_user_from_cookie(session_token)
 302          if row is None:
 303              raise HTTPException(401, "Non authentifie")
 304          fields = {k: v for k, v in payload.dict(exclude_unset=True).items() if v is not None}
 305          if not fields:
 306              return _public(row)
 307          sets = ", ".join(k + " = ?" for k in fields.keys())
 308          vals = list(fields.values())
 309          conn = get_db()
 310          try:
 311              conn.execute(
 312                  "UPDATE users SET " + sets + ", updated_at = ? WHERE id = ?",
 313                  vals + [datetime.now().isoformat(timespec="seconds"), row["id"]],
 314              )
 315              conn.commit()
 316              updated = conn.execute("SELECT * FROM users WHERE id = ?", (row["id"],)).fetchone()
 317          finally:
 318              conn.close()
 319          return _public(updated)

 374      @app.post("/api/auth/reset-password")
 375      def reset_password(payload: ResetRequest):
 376          token = payload.token.strip()
 377          pwd = payload.new_password
 378          if not token or len(pwd) < 6:
 379              raise HTTPException(400, "Mot de passe trop court (6 caracteres minimum)")
 380  
 381          conn = get_db()
 382          try:
 383              row = conn.execute(
 384                  "SELECT * FROM password_resets WHERE token = ? AND used = 0",
 385                  (token,)
 386              ).fetchone()
 387              if row is None:
 388                  raise HTTPException(400, "Lien invalide ou deja utilise")
 389              try:
 390                  exp = datetime.fromisoformat(row["expires_at"])
 391                  if datetime.now() > exp:
 392                      raise HTTPException(400, "Lien expire")
 393              except ValueError:
 394                  pass
 395  
 396              try:
 397                  pwd_hash = hash_password(pwd)
 398              except ValueError as e:
 399                  raise HTTPException(400, str(e))
 400  
 401              conn.execute(
 402                  "UPDATE users SET password_hash=?, updated_at=? WHERE id=?",
 403                  (pwd_hash, datetime.now().isoformat(timespec="seconds"), row["user_id"])
 404              )
 405              conn.execute(
 406                  "UPDATE password_resets SET used=1 WHERE id=?",
 407                  (row["id"],)
 408              )
 409              # Invalider toutes les sessions de cet utilisateur
 410              conn.execute("DELETE FROM sessions WHERE user_id=?", (row["user_id"],))
 411              conn.commit()
 412          finally:
 413              conn.close()
 414  
 415          return {"ok": True, "message": "Mot de passe reinitialise. Vous pouvez vous connecter."}
